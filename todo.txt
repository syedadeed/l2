CREATE TYPE BUSINESS_TAG AS ENUM (...);

CREATE TABLE business_users (
    id                  UUID            PRIMARY KEY,
    username            TEXT            NOT NULL UNIQUE CHECK(username ~ '^[A-Za-z0-9_-]+$'),
    created_at          TIMESTAMPTZ     NOT NULL DEFAULT NOW(),
    is_active           BOOLEAN         NOT NULL DEFAULT TRUE,
    is_verified         BOOLEAN         NOT NULL DEFAULT FALSE,
    name                TEXT,
    upi_id              TEXT,
    country             TEXT,
    state               TEXT,
    city                TEXT,
    postal_code         TEXT,
    street_address      TEXT,
    extended_address    TEXT,
    location            POINT,
    timezone            TEXT,
    banner_key          UUID,
    avatar_key          UUID,
    tagline             TEXT,
    bio                 TEXT,
    phone_country_code  TEXT,
    phone_number        TEXT,
    email               TEXT,
    website_url         TEXT,
    instagram_url       TEXT,
    x_url               TEXT,
    youtube_url         TEXT,
    facebook_url        TEXT,
    tags                BUSINESS_TAG[]  NOT NULL DEFAULT ARRAY[]::business_tag[]
);

CREATE TABLE business_hours (
    id          UUID PRIMARY KEY,
    business_id UUID NOT NULL REFERENCES business_users(id) ON DELETE CASCADE,
    day_range   INT4RANGE NOT NULL CHECK (NOT isempty(day_range) AND day_range <@ int4range(0, 7)),
    open_time   TIME NOT NULL,
    close_time  TIME NOT NULL,
    CHECK (open_time < close_time)
);
