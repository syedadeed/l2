CREATE TABLE business_users (
    id               UUID  PRIMARY KEY,
    username         TEXT  NOT NULL UNIQUE CHECK(username ~ '^[A-Za-z0-9_-]+$'),
    name             TEXT,
    upi_id           TEXT  CHECK(upi_id ~ '^[A-Za-z0-9._-]+@[A-Za-z0-9._-]+$'),
    gst_number       TEXT  CHECK(gst_number ~ '^[A-Z0-9]{15}$'),
    country          TEXT,
    state            TEXT,
    city             TEXT,
    postal_code      TEXT,
    street_address   TEXT,
    extended_address TEXT,
    location         POINT CHECK(location[0] BETWEEN -180 AND 180 AND location[1] BETWEEN -90 AND 90),
    banner_key       UUID,
    avatar_key       UUID,
    tagline          TEXT,
    bio              TEXT,
);

CREATE validate_location_sync() RETURNS TRIGGER LANGUAGE PLPGSQL AS
$$
BEGIN
    IF (NEW.country IS DISTINCT FROM OLD.country OR
        NEW.state IS DISTINCT FROM OLD.state OR
        NEW.city IS DISTINCT FROM OLD.city OR
        NEW.postal_code IS DISTINCT FROM OLD.postal_code OR
        NEW.street_address IS DISTINCT FROM OLD.street_address)
       <>
       (NEW.location IS DISTINCT FROM OLD.location)
    THEN
        RAISE EXCEPTION 'location and address fields must be updated together';
    END IF;

    RETURN NEW;
END;
$$;

CREATE TRIGGER business_users_location_sync
BEFORE UPDATE ON business_users
FOR EACH ROW
EXECUTE FUNCTION validate_location_sync();
